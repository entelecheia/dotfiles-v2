# Default include rules for `dot sync` include mode (the union filter's
# untracked-binary half).
#
# Include mode is the default strategy: git-tracked files always sync (via
# tracked-includes.dyn.conf), and untracked files sync only when they match
# these binary/artifact patterns. Matching is case-insensitive in both
# Go-side planning and generated rsync rules.

# archives
*.zip
*.tar
*.tgz
*.gz
*.rar
*.zst

# audio
*.ogg
*.mp3
*.wav
*.flac

# video / subtitles
*.mp4
*.avi
*.mov
*.mkv
*.srt

# images / design
*.png
*.jpg
*.jpeg
*.heic
*.webp
*.bmp
*.wmf
*.ai

# documents / decks / sheets
*.key
*.pdf
*.hwp*
*.doc
*.docx
*.ppt
*.pptx
*.ppsx
*.pps
*.xls*
*.xlsx
*.xlsm

# data / ML artifacts (absorbed from the retired binary-extensions.conf)
*.parquet
*.pkl
*.pt

*.tsv
